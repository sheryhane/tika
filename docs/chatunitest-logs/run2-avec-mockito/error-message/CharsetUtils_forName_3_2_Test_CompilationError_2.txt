@ExtendWith(MockitoExtension.class)
public class CharsetUtils_forName_3_2_Test {

    @Mock
    private static Method getCharsetICU;

    @Mock
    private static Method isSupportedICU;

    private CharsetUtils charsetUtils;

    @BeforeEach
    public void setUp() throws Exception {
        charsetUtils = new CharsetUtils();
        commonCharsets = new HashMap<>();
        commonCharsets.put("utf-8", Charset.forName("UTF-8"));
        commonCharsets.put("iso-8859-1", Charset.forName("ISO-8859-1"));
    }

    @Test
    public void testForName_withNullName() {
        assertThrows(IllegalArgumentException.class, () -> charsetUtils.forName(null));
    }

    @Test
    public void testForName_withValidName() throws Exception {
        when(getCharsetICU.invoke(null, "UTF-8")).thenReturn(null);
        Charset charset = charsetUtils.forName("UTF-8");
        assertEquals(Charset.forName("UTF-8"), charset);
    }

    @Test
    public void testForName_withInvalidName() {
        assertThrows(IllegalCharsetNameException.class, () -> charsetUtils.forName("invalid-name"));
    }

    @Test
    public void testForName_withISOName() throws Exception {
        when(getCharsetICU.invoke(null, "iso-8859-1")).thenReturn(null);
        Charset charset = charsetUtils.forName("iso 8859-1");
        assertEquals(Charset.forName("ISO-8859-1"), charset);
    }

    @Test
    public void testForName_withCPName() throws Exception {
        when(getCharsetICU.invoke(null, "cp1252")).thenReturn(null);
        Charset charset = charsetUtils.forName("cp-1252");
        assertEquals(Charset.forName("cp1252"), charset);
    }

    @Test
    public void testForName_withWinName() throws Exception {
        when(getCharsetICU.invoke(null, "windows-1252")).thenReturn(null);
        Charset charset = charsetUtils.forName("win-1252");
        assertEquals(Charset.forName("windows-1252"), charset);
    }

    @Test
    public void testForName_withNoneName() {
        assertThrows(IllegalCharsetNameException.class, () -> charsetUtils.forName("none"));
    }

    @Test
    public void testForName_withNoName() {
        assertThrows(IllegalCharsetNameException.class, () -> charsetUtils.forName("no"));
    }

    @Test
    public void testForName_withICUCharset() throws Exception {
        when(getCharsetICU.invoke(null, "UTF-8")).thenReturn(Charset.forName("UTF-8"));
        Charset charset = charsetUtils.forName("UTF-8");
        assertEquals(Charset.forName("UTF-8"), charset);
    }

    @Test
    public void testForName_withICUUnsupportedCharset() throws Exception {
        when(getCharsetICU.invoke(null, "invalid-name")).thenReturn(null);
        when(isSupportedICU.invoke(null, "invalid-name")).thenReturn(false);
        assertThrows(IllegalCharsetNameException.class, () -> charsetUtils.forName("invalid-name"));
    }
}
